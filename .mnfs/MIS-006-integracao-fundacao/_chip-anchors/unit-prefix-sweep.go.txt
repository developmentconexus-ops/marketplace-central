package main

// Class sweep for CHIP-ANCHORS: "a unit token matches as a PREFIX of an ordinary
// Portuguese word that follows a digit, fabricating a measurement that is not one."
// Same mechanism as the bare-`in` defect. This enumerates every unit token in the
// shipped alternation against realistic PT-BR retail vocabulary.

import (
	"fmt"
	"regexp"
)

// Verbatim from generation_service.go:655 at commit f92ca9c7.
var shipped = regexp.MustCompile(`(?i)\d+\s*x\s*\d+(?:\s*x\s*\d+)?|\d+(?:[.,]\d+)?\s*(?:mm|cm|pol|")|\d+(?:[.,]\d+)?\s*m\b`)

type probe struct {
	unit  string
	title string
	note  string
}

func main() {
	probes := []probe{
		{"mm", "Kit 30 mmhg meia compressao", "no common PT word starts 'mm'; contrived"},
		{"cm", "Regua 30 cmyk impressao", "no common PT word starts 'cm'; contrived"},
		{"pol", "Blusa 30 poliester 70 algodao", "fabric composition — REAL retail phrasing"},
		{"pol", "Suporte 2 polias de aco", "polia = pulley — REAL hardware phrasing"},
		{"pol", "Toalheiro 500MM CR/POLIDO", "the golden ERP string"},
		{"pol", "Monitor 24 polegadas full hd", "polegadas = inches — a TRUE measurement"},
		{"pol", "Broca 2 pol aco rapido", "bare 'pol' — a TRUE measurement"},
		{"m", "Cabo 5 metros flexivel", "bare m has \\b in the pattern"},
		{"m", "Camiseta 3 malhas algodao", "bare m followed by a word"},
		{"\"", "TV 32\" smart", "inch mark"},
	}

	fmt.Printf("%-5s  %-34s  %-22s  %s\n", "UNIT", "TITLE", "MATCHES", "VERDICT")
	for _, p := range probes {
		found := shipped.FindAllString(p.title, -1)
		verdict := "clean"
		if len(found) > 0 {
			verdict = "matched"
		}
		fmt.Printf("%-5s  %-34s  %-22s  %s  (%s)\n", p.unit, p.title, fmt.Sprint(found), verdict, p.note)
	}

	// The candidate fix, applied to `pol` only: spell the long forms out and
	// require a word boundary after the short one.
	fixed := regexp.MustCompile(`(?i)\d+\s*x\s*\d+(?:\s*x\s*\d+)?|\d+(?:[.,]\d+)?\s*(?:mm|cm|polegadas|polegada|pol\b|")|\d+(?:[.,]\d+)?\s*m\b`)
	fmt.Println("\n--- candidate fix: (?:mm|cm|polegadas|polegada|pol\\b|\") ---")
	for _, p := range probes {
		if p.unit != "pol" {
			continue
		}
		fmt.Printf("%-34s  shipped=%-14s  fixed=%s\n", p.title, fmt.Sprint(shipped.FindAllString(p.title, -1)), fmt.Sprint(fixed.FindAllString(p.title, -1)))
	}
}
